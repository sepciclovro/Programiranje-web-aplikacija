    lastname VARCHAR(50) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    username VARCHAR(10) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    country VARCHAR(100) NOT NULL

<?php
$conn = mysqli_connect("localhost", "root", "", "vjezba_16");

if (!$conn) {
    die("Došlo je do pogreške pri radu s bazom podataka: " . mysqli_connect_error());
}

mysqli_set_charset($conn, "utf8mb4");

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $firstname = trim($_POST['firstname']);
    $lastname = trim($_POST['lastname']);
    $email = trim($_POST['email']);
    $username = trim($_POST['username']);
    $password = $_POST['password'];
    $country = $_POST['country'];

    if (strlen($username) < 5 || strlen($username) > 10) {
        die("Korisničko ime mora imati između 5 i 10 znakova.");
    }

    if (strlen($password) < 4) {
        die("Lozinka mora imati minimalno 4 znaka.");
    }

    $query_check = "SELECT id FROM users WHERE username = ? OR email = ?";
    $stmt_check = mysqli_prepare($conn, $query_check);
    mysqli_stmt_bind_param($stmt_check, "ss", $username, $email);
    mysqli_stmt_execute($stmt_check);
    mysqli_stmt_store_result($stmt_check);

    if (mysqli_stmt_num_rows($stmt_check) > 0) {
        echo "<h3>Korisničko ime ili email se već koriste.</h3>";
        mysqli_stmt_close($stmt_check);
    } else {
        mysqli_stmt_close($stmt_check);

        $hashed_password = password_hash($password, PASSWORD_DEFAULT);

        $query_insert = "INSERT INTO users (firstname, lastname, email, username, password, country) VALUES (?, ?, ?, ?, ?, ?)";
        $stmt_insert = mysqli_prepare($conn, $query_insert);
        
        mysqli_stmt_bind_param($stmt_insert, "ssssss", $firstname, $lastname, $email, $username, $hashed_password, $country);

        if (mysqli_stmt_execute($stmt_insert)) {
            echo "<h3>Registracija je uspješno završena!</h3>";
        } else {
            echo "Došlo je do pogreške prilikom upisa: " . mysqli_error($conn);
        }

        mysqli_stmt_close($stmt_insert);
    }
}

mysqli_close($conn);
?>
