CREATE DATABASE IF NOT EXISTS vjezba_14 DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE vjezba_14;

CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    ime VARCHAR(50) NOT NULL,
    prezime VARCHAR(50) NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    lozinka VARCHAR(255) NOT NULL,
    drzava VARCHAR(100) NOT NULL,
    opis_korisnika TEXT,
    vrijeme_zapisa TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);
