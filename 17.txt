CREATE DATABASE IF NOT EXISTS vjezba_17 DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE vjezba_17;

CREATE TABLE IF NOT EXISTS countries (
    id INT AUTO_INCREMENT PRIMARY KEY,
    country_name VARCHAR(100) NOT NULL
);

CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    firstname VARCHAR(50) NOT NULL,
    lastname VARCHAR(50) NOT NULL,
    country_id INT,
    CONSTRAINT fk_user_country FOREIGN KEY (country_id) REFERENCES countries(id) ON DELETE SET NULL
);

<?php
$conn = mysqli_connect("localhost", "root", "", "vjezba_17");

if (!$conn) {
    die("Došlo je do pogreške pri radu s bazom podataka: " . mysqli_connect_error());
}

mysqli_set_charset($conn, "utf8mb4");

$query = "
    SELECT users.firstname, users.lastname, countries.country_name 
    FROM users 
    INNER JOIN countries ON users.country_id = countries.id
";

$result = mysqli_query($conn, $query);

if (mysqli_num_rows($result) > 0) {
    echo "<ul>";
    while ($row = mysqli_fetch_assoc($result)) {
        $fullname = htmlspecialchars($row['firstname'] . " " . $row['lastname']);
        $country = htmlspecialchars($row['country_name']);
        echo "<li>" . $fullname . " (" . $country . ")</li>";
    }
    echo "</ul>";
} else {
    echo "<p>Nema pronađenih korisnika.</p>";
}

mysqli_close($conn);
?>
